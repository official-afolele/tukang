Fungsi Buka Dokumen dari ID function loadScribdById() { let inputVal = document.getElementById('scribd-id-input').value.trim(); let match = inputVal.match(/\d{6,}/); let docId = match ? match[0] : inputVal; if (!docId) { alert("Silakan masukkan ID dokumen terlebih dahulu!"); document.getElementById('scribd-id-input').focus(); return; } const fullUrl = `https://www.scribd.com/embeds/${docId}/content?start_page=1&view_mode=scroll&access_key=key-fFexxf7r1bzEfWu3HKwf`; document.getElementById('scribd-frame').src = fullUrl; } // 3. Fungsi Simpan PDF Langsung (Tanpa Meninggalkan Tab Baru) function downloadAsPDF() { const frame = document.getElementById('scribd-frame'); try { // Coba print langsung di dalam frame halaman saat ini frame.contentWindow.focus(); frame.contentWindow.print(); } catch (err) { // Jika dibatasi izin browser, gunakan background printer yang langsung tertutup otomatis let inputVal = document.getElementById('scribd-id-input').value.trim(); let match = inputVal.match(/\d{6,}/); let docId = match ? match[0] : inputVal; const fullUrl = `https://www.scribd.com/embeds/${docId}/content?start_page=1&view_mode=scroll&access_key=key-fFexxf7r1bzEfWu3HKwf`; // Membuka jendela dialog simpan PDF dan langsung menutup dirinya sendiri saat selesai const printWindow = window.open(fullUrl, '_blank', 'width=800,height=600'); if (printWindow) { printWindow.onload = function() { printWindow.focus(); printWindow.print(); // Otomatis tertutup sendiri begitu Anda klik Save / Batal printWindow.onafterprint = function() { printWindow.close(); }; }; } } }
